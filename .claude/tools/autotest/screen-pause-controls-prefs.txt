# CONTROLS FROM THE PAUSE MENU (2026-11-10), the two buttons it shares with its neighbours:
# - Y is DONE on Preferences, so it comes back to the pause menu still held: that Y must NOT open the list.
# - START is ESC while the pause menu is up (gamepad_ui.cc), so it closes the list -- and only the list.
name screen-pause-controls-prefs-f2
map DESERT1.MAP
seed 1
duration 16
dude tile 20100 rot 2

# START tapped: the pause menu opens on the release, its highlight on SAVE GAME
open 1000 options
# SAVE -> LOAD -> PREFERENCES, A opens it
tap 2200 DOWN
tap 2700 DOWN
tap 3200 A
# Y = DONE on Preferences, held 0.5 s past the return to the pause menu
tap 4600 Y 500
shot 5600
# a fresh Y opens the list, START closes it again
tap 6400 Y
shot 7400
tap 8200 START
shot 9000
# B shuts the pause menu
tap 10000 B
note 12000 the world should be back

# the checks (run.js)
expect-count 1 \[FLOW/controls-open\]
expect \[FLOW/controls-open\] rows=24 size=\d+x\d+
expect \[FLOW/controls-closed\] by=esc
# the pause menu stayed up from START to the last B
expect-not \[FLOW/autotest\] modal none t=[2-9]\d\d\d
expect \[FLOW/autotest\] end why=time .* modal=-1
