# SQUAD COMMAND MODE by the pad: its buttons (2026-11-12, his design "A"). LT + A opens the mode (the world slows,
# then holds); A on free ground is no order since GO went (2026-11-13): the buzz, and the mode stays; B closes it with
# no order; LT + A then X gives HOLD FIRE to All and closes it; LT + A, RB (Sulik, the nearest), RB (Vic), D-pad Down
# gives FOLLOW to Vic alone and closes it; LT + A then Y (his pick of 2026-11-13) gives NO HEALING to All and closes it.
# The opening A is the mode's: no interaction, no loot, no use-on cursor. The cursor's orders by the pad are
# squad-pad-heal, -weapon, -attack, -holdfire, -noheal; the slowdown and the hold are squad-mode-slow; the pictures are
# screen-squad-mode and squad-pad-attack.
# Vic (pistol) 4 tiles left of the dude; Sulik (knife) beside him; no enemy, so nothing moves the world but the orders.
#   node ../fallout1-ce/.claude/tools/autotest/run.js .claude/tools/autotest/squad-mode-pad.txt 1 --exe build/Release/fallout2-ce.exe
name squad-mode-pad
map DESERT1.MAP
seed 1
duration 14
dude tile 20100 rot 2
dude hp 400
dude curhp 200
dude armor 2

spawn 16777278 19904 rot 5 wield 8 item 29 3 item 40 2 party
spawn 16777313 20300 rot 4 wield 4 party

# 1. LT + A: the mode opens, the cursor on free ground 3 tiles ahead; A there: no order, the buzz, the mode stays; B (in
#    the slowdown): it closes with no order.
axis 500 LT 32767
tap 700 A
tap 1100 A
tap 1500 B
axis 1800 LT -32768
# 2. LT + A, X (after the hold): HOLD FIRE to All; the order closes the mode.
axis 2500 LT 32767
tap 2700 A
tap 4600 X
axis 4900 LT -32768
# 3. LT + A, RB (who = the nearest companion, Sulik), RB (Vic), D-pad Down: FOLLOW to Vic alone; closes.
axis 5500 LT 32767
tap 5700 A
tap 6100 RB
tap 6400 RB
tap 6800 DOWN
axis 7100 LT -32768
# 4. LT + A, Y (after the hold): NO HEALING to All; the order closes the mode.
axis 8000 LT 32767
tap 8200 A
tap 10100 Y
axis 10400 LT -32768
note 12500 end

# the checks (run.js)
expect-count 4 \[FLOW/squad-mode\] open why=lt\+a slow=1\.50
expect \[INPUT/squad-a\] kind=none order=- obj=-1 tile=\d+ who=all ok=0
expect \[FLOW/squad-mode\] close why=b slowed=0\.\d\d frozen=0
expect \[FLOW/squad-mode\] close why=x slowed=1\.5\d frozen=1 heldMs=\d+ rtHeld=0\.000
expect \[ACTION/squad-order\] order=hold-fire target=- companions=2 skipped=0 why=ok speaker=\d+ t=[\d.]+ who=all
expect \[INPUT/squad-who\] who=\w+ id=18097 pid=16777313 index=0 of=2
expect \[INPUT/squad-who\] who=\w+ id=18062 pid=16777278 index=1 of=2
expect \[INPUT/squad-dpad\] dir=down order=follow who=18062 ok=1
expect \[ACTION/squad-order\] order=follow target=- companions=1 skipped=0 why=ok speaker=18062 t=[\d.]+ who=18062
expect \[FLOW/squad-mode\] close why=dpad
expect \[INPUT/squad-y\] order=no-healing who=all ok=1
expect \[ACTION/squad-order\] order=no-healing target=- companions=2 skipped=0 why=ok speaker=\d+ t=[\d.]+ who=all
expect \[FLOW/medic-optout\] id=18062 treats=0
expect \[FLOW/medic-optout\] id=18097 treats=0
expect \[FLOW/squad-mode\] close why=y slowed=1\.5\d frozen=1 heldMs=\d+ rtHeld=0\.000
# A on free ground was refused (the mode stayed for B), and nothing of a GO is left
expect-count 1 \[INPUT/squad-mode\] close why=b
expect-not \[INPUT/squad-a\] kind=ground
# the opening A is spent: no interaction (interact_prompt's request / its walk-then-act), no world click
expect-not \[ACTION/interact-(queue|do)\]
expect-not \[INPUT/key\] code=-2
