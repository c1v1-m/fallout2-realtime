#   3 Y: HEAL AGAIN -- the mode stays; "I'll patch you up." over Vic again, the legend back to "Y No healing"
#   4 B: the mode closed

# the checks (run.js)
expect-count 1 \[INPUT/squad-mode\] open members=2 start=ahead x=474 y=226 tile=20700 kind=ground
expect-count 1 \[INPUT/squad-cursor\] on kind=dude ok=1
expect \[INPUT/squad-y\] order=no-healing who=all ok=1
expect \[ACTION/squad-order\] order=no-healing target=- companions=2 skipped=0 why=ok speaker=18062 t=[\d.]+ who=all
expect \[FLOW/medic-optout\] id=18062 treats=0
expect \[FLOW/medic-optout\] id=18097 treats=0
expect \[INPUT/squad-y\] order=heal-again who=all ok=1
expect \[ACTION/squad-order\] order=heal-again target=- companions=2 skipped=0 why=ok speaker=18062 t=[\d.]+ who=all
expect \[FLOW/medic-optout\] id=18062 treats=1
expect \[FLOW/medic-optout\] id=18097 treats=1
# ...and the list is empty again (nearest first: Sulik, then Vic)
expect \[FLOW/medic-optout\] id=\d+ treats=1 count=0
# both Ys click and leave the mode open; B alone closes it
expect \[INPUT/squad-mode\] stays after=y order=no-healing speaker=18062
expect \[INPUT/squad-mode\] stays after=y order=heal-again speaker=18062
expect-count 2 \[INPUT/squad-mode\] stays after=
expect-count 1 \[INPUT/squad-mode\] close why=
expect \[INPUT/squad-mode\] close why=b
# the Y was the mode's both times: no LT + Y firing mode, no Y-hold weapon wheel, no interaction, no other order
expect-not \[INPUT/fire-mode\]
expect-not \[FLOW/world-hold\] on why=wheel
expect-not \[ACTION/interact-(queue|do)\]
expect-not \[ACTION/squad-order\] order=(?!no-healing|heal-again)
